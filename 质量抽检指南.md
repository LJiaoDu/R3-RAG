# Reference Answer 质量抽检指南

## 为什么需要抽检？

即使使用了改进的Prompt，仍然可能出现问题：
- ❌ GPT-4偶尔会"幻觉"
- ❌ 检索结果可能不准确
- ❌ Citation可能无效
- ❌ Facts可能与文档不符

**抽检目标**：在数据进入训练前，发现并修复问题

---

## 完整抽检流程

```
生成7000个答案
  ↓
[步骤1] 自动化检查（100%）
  - 结构完整性
  - 引用有效性
  - 基本质量指标
  ↓
[步骤2] 分层抽样（5-10%）
  - 随机抽样
  - 重点抽样（低质量、高unknown等）
  ↓
[步骤3] 人工检查
  - 事实准确性
  - 引用正确性
  - 完整性评估
  ↓
[步骤4] 问题修复
  - 重新生成
  - 手动修正
  ↓
[步骤5] 质量报告
  - 统计分析
  - 问题总结
  - 改进建议
```

## 步骤1: 自动化检查（100%覆盖）

### 检查脚本

```python
#!/usr/bin/env python3
"""
自动化质量检查脚本
对所有生成的Reference Answer进行自动检查
"""

import json
from typing import List, Dict
from collections import defaultdict

def load_reference_answers(file_path: str) -> List[Dict]:
    """加载所有Reference Answers"""
    results = []
    with open(file_path, 'r', encoding='utf-8') as f:
        for line in f:
            data = json.loads(line.strip())
            results.append(data)
    return results


def automatic_check_all(results: List[Dict]) -> Dict:
    """
    对所有结果进行自动检查

    返回检查报告
    """

    report = {
        'total': len(results),
        'issues': defaultdict(list),
        'statistics': {}
    }

    for i, result in enumerate(results):
        question = result.get('question', f'Question_{i}')

        # 检查1: 结构完整性
        if not result.get('success'):
            report['issues']['failed_generation'].append({
                'index': i,
                'question': question,
                'error': result.get('error', 'Unknown error')
            })
            continue

        structured = result.get('structured_answer')
        if not structured:
            report['issues']['no_structured_answer'].append({
                'index': i,
                'question': question
            })
            continue

        # 检查2: 必需字段
        required_fields = ['answer_summary', 'facts', 'unknown']
        missing_fields = [f for f in required_fields if f not in structured]
        if missing_fields:
            report['issues']['missing_fields'].append({
                'index': i,
                'question': question,
                'missing': missing_fields
            })

        # 检查3: 引用完整性
        facts = structured.get('facts', [])
        for j, fact in enumerate(facts):
            if 'citations' not in fact or len(fact['citations']) == 0:
                report['issues']['missing_citations'].append({
                    'index': i,
                    'question': question,
                    'fact_index': j,
                    'claim': fact.get('claim', 'Unknown')
                })

        # 检查4: 引用有效性
        invalid_citations = result.get('invalid_citations', [])
        if len(invalid_citations) > 0:
            report['issues']['invalid_citations'].append({
                'index': i,
                'question': question,
                'invalid': invalid_citations
            })

        # 检查5: 质量分数
        quality_score = result.get('quality_score', 0)
        if quality_score < 0.6:
            report['issues']['low_quality'].append({
                'index': i,
                'question': question,
                'score': quality_score
            })

        # 检查6: Facts为空
        if len(facts) == 0:
            report['issues']['no_facts'].append({
                'index': i,
                'question': question
            })

        # 检查7: Answer summary为空
        answer_summary = structured.get('answer_summary', '')
        if len(answer_summary.strip()) < 10:
            report['issues']['empty_answer'].append({
                'index': i,
                'question': question
            })

    # 统计信息
    report['statistics'] = {
        'success_rate': sum(1 for r in results if r.get('success')) / len(results),
        'avg_quality_score': sum(r.get('quality_score', 0) for r in results) / len(results),
        'avg_facts_count': sum(len(r.get('structured_answer', {}).get('facts', [])) for r in results) / len(results),
        'avg_unknown_count': sum(len(r.get('structured_answer', {}).get('unknown', [])) for r in results) / len(results),
        'questions_with_issues': sum(len(v) for v in report['issues'].values())
    }

    return report


def print_report(report: Dict):
    """打印检查报告"""

    print(f"\n{'='*60}")
    print(f"自动化检查报告")
    print(f"{'='*60}")

    print(f"\n📊 统计信息:")
    print(f"  总数: {report['total']}")
    print(f"  成功率: {report['statistics']['success_rate']:.1%}")
    print(f"  平均质量分: {report['statistics']['avg_quality_score']:.2f}")
    print(f"  平均Facts数: {report['statistics']['avg_facts_count']:.1f}")
    print(f"  平均Unknown数: {report['statistics']['avg_unknown_count']:.1f}")
    print(f"  有问题的数量: {report['statistics']['questions_with_issues']}")

    print(f"\n🔍 问题详情:")

    for issue_type, issues in report['issues'].items():
        if len(issues) > 0:
            print(f"\n  [{issue_type}] {len(issues)}个问题")
            # 显示前3个示例
            for issue in issues[:3]:
                print(f"    - 问题 {issue['index']}: {issue['question'][:50]}...")
            if len(issues) > 3:
                print(f"    ... 还有 {len(issues) - 3} 个")

    print(f"\n{'='*60}")


def save_report(report: Dict, output_file: str):
    """保存详细报告"""
    with open(output_file, 'w', encoding='utf-8') as f:
        json.dump(report, f, indent=2, ensure_ascii=False)
    print(f"\n详细报告已保存到: {output_file}")


# 使用示例
if __name__ == "__main__":
    # 加载数据
    results = load_reference_answers('reference_answers.jsonl')

    # 自动检查
    report = automatic_check_all(results)

    # 打印报告
    print_report(report)

    # 保存详细报告
    save_report(report, 'quality_check_report.json')
```

## 步骤2: 分层抽样（5-10%）

### 抽样策略

```python
#!/usr/bin/env python3
"""
分层抽样策略
"""

import random
from typing import List, Dict

def stratified_sampling(results: List[Dict], sample_size: int = 500) -> Dict[str, List[Dict]]:
    """
    分层抽样

    分层：
    1. 随机抽样（基线）
    2. 低质量样本（重点检查）
    3. 高unknown样本（可能信息不足）
    4. 复杂问题（多facts）
    5. 简单问题（少facts）
    """

    samples = {
        'random': [],
        'low_quality': [],
        'high_unknown': [],
        'complex': [],
        'simple': []
    }

    # 1. 随机抽样（30%）
    random_sample_size = int(sample_size * 0.3)
    samples['random'] = random.sample(results, random_sample_size)

    # 2. 低质量样本（30%）
    low_quality = sorted(results, key=lambda x: x.get('quality_score', 1.0))
    samples['low_quality'] = low_quality[:int(sample_size * 0.3)]

    # 3. 高unknown样本（20%）
    high_unknown = sorted(
        results,
        key=lambda x: len(x.get('structured_answer', {}).get('unknown', [])),
        reverse=True
    )
    samples['high_unknown'] = high_unknown[:int(sample_size * 0.2)]

    # 4. 复杂问题（10%）
    complex_qs = sorted(
        results,
        key=lambda x: len(x.get('structured_answer', {}).get('facts', [])),
        reverse=True
    )
    samples['complex'] = complex_qs[:int(sample_size * 0.1)]

    # 5. 简单问题（10%）
    simple_qs = sorted(
        results,
        key=lambda x: len(x.get('structured_answer', {}).get('facts', []))
    )
    samples['simple'] = simple_qs[:int(sample_size * 0.1)]

    return samples


def save_samples_for_review(samples: Dict[str, List[Dict]], output_dir: str = 'review'):
    """保存抽样结果供人工检查"""
    import os
    os.makedirs(output_dir, exist_ok=True)

    for category, sample_list in samples.items():
        output_file = f"{output_dir}/{category}_samples.jsonl"
        with open(output_file, 'w', encoding='utf-8') as f:
            for item in sample_list:
                f.write(json.dumps(item, ensure_ascii=False) + '\n')

        print(f"✅ {category}: {len(sample_list)}个样本 → {output_file}")


# 使用示例
if __name__ == "__main__":
    results = load_reference_answers('reference_answers.jsonl')

    # 抽样（假设7000个，抽500个 ≈ 7%）
    samples = stratified_sampling(results, sample_size=500)

    # 保存
    save_samples_for_review(samples)

    print(f"\n总抽样数: {sum(len(v) for v in samples.values())}")
```

## 步骤4: 引用验证工具

```python
#!/usr/bin/env python3
"""
引用验证工具 - 验证每个citation是否真实支持claim
"""

def verify_claim_against_citation(claim: str, citation_id: str, retrieved_docs: List[Dict]) -> Dict:
    """
    验证claim是否被citation支持

    返回:
    {
        'valid': bool,
        'citation_content': str,
        'match_score': float,
        'issues': []
    }
    """

    # 1. 提取citation对应的chunk内容
    doc_id, chunk_id = citation_id.split('#')

    chunk_content = None
    for doc in retrieved_docs:
        if doc['doc_id'] == doc_id:
            if 'chunks' in doc:
                for chunk in doc['chunks']:
                    if chunk['chunk_id'] == chunk_id:
                        chunk_content = chunk['content']
                        break
            else:
                chunk_content = doc.get('content', '')
            break

    if chunk_content is None:
        return {
            'valid': False,
            'citation_content': None,
            'match_score': 0.0,
            'issues': ['Citation不存在']
        }

    # 2. 提取claim中的关键信息
    import re

    # 课程代码
    claim_codes = set(re.findall(r'\b[A-Z]{2,3}\d{4}\b', claim))
    content_codes = set(re.findall(r'\b[A-Z]{2,3}\d{4}\b', chunk_content))

    # ECTS
    claim_ects = set(re.findall(r'(\d+)\s*ECTS', claim, re.IGNORECASE))
    content_ects = set(re.findall(r'(\d+)\s*ECTS', chunk_content, re.IGNORECASE))

    # 关键词
    claim_keywords = set(claim.lower().split())
    content_keywords = set(chunk_content.lower().split())

    # 3. 计算匹配度
    issues = []
    match_score = 0.0

    # 课程代码匹配
    if claim_codes:
        code_match = len(claim_codes & content_codes) / len(claim_codes)
        match_score += code_match * 0.4
        if code_match < 1.0:
            issues.append(f"课程代码不匹配: claim有{claim_codes}, content有{content_codes}")

    # ECTS匹配
    if claim_ects:
        ects_match = len(claim_ects & content_ects) / len(claim_ects)
        match_score += ects_match * 0.3
        if ects_match < 1.0:
            issues.append(f"ECTS不匹配: claim有{claim_ects}, content有{content_ects}")

    # 关键词匹配
    keyword_match = len(claim_keywords & content_keywords) / len(claim_keywords)
    match_score += keyword_match * 0.3

    # 4. 判断
    valid = match_score >= 0.6 and len(issues) == 0

    return {
        'valid': valid,
        'citation_content': chunk_content,
        'match_score': match_score,
        'issues': issues
    }


def batch_verify_citations(results: List[Dict]) -> Dict:
    """批量验证所有citations"""

    report = {
        'total_facts': 0,
        'total_citations': 0,
        'valid_citations': 0,
        'invalid_citations': [],
        'suspicious_citations': []
    }

    for i, result in enumerate(results):
        if not result.get('success'):
            continue

        structured = result['structured_answer']
        facts = structured.get('facts', [])
        retrieved_docs = result.get('retrieved_docs', [])

        for fact in facts:
            report['total_facts'] += 1
            claim = fact['claim']
            citations = fact.get('citations', [])

            for citation in citations:
                report['total_citations'] += 1

                verification = verify_claim_against_citation(
                    claim, citation, retrieved_docs
                )

                if verification['valid']:
                    report['valid_citations'] += 1
                elif verification['match_score'] >= 0.4:
                    # 可疑但不完全无效
                    report['suspicious_citations'].append({
                        'question': result['question'],
                        'claim': claim,
                        'citation': citation,
                        'score': verification['match_score'],
                        'issues': verification['issues']
                    })
                else:
                    # 完全无效
                    report['invalid_citations'].append({
                        'question': result['question'],
                        'claim': claim,
                        'citation': citation,
                        'score': verification['match_score'],
                        'issues': verification['issues']
                    })

    report['valid_rate'] = report['valid_citations'] / report['total_citations'] if report['total_citations'] > 0 else 0

    return report


# 使用示例
if __name__ == "__main__":
    results = load_reference_answers('reference_answers.jsonl')

    print("开始验证citations...")
    report = batch_verify_citations(results)

    print(f"\n{'='*60}")
    print(f"Citation验证报告")
    print(f"{'='*60}")
    print(f"总Facts数: {report['total_facts']}")
    print(f"总Citations数: {report['total_citations']}")
    print(f"有效Citations: {report['valid_citations']} ({report['valid_rate']:.1%})")
    print(f"无效Citations: {len(report['invalid_citations'])}")
    print(f"可疑Citations: {len(report['suspicious_citations'])}")

    if report['invalid_citations']:
        print(f"\n前3个无效citations:")
        for item in report['invalid_citations'][:3]:
            print(f"  问题: {item['question']}")
            print(f"  Claim: {item['claim']}")
            print(f"  Citation: {item['citation']}")
            print(f"  问题: {item['issues']}")
            print()
```

```python
#!/usr/bin/env python3
"""
综合质量报告
"""

def generate_quality_report(results: List[Dict], manual_review: Dict = None):
    """
    生成综合质量报告
    """

    report = {
        'overview': {},
        'quality_distribution': {},
        'common_issues': {},
        'recommendations': []
    }

    # 1. 概览统计
    total = len(results)
    successful = sum(1 for r in results if r.get('success'))

    report['overview'] = {
        'total_questions': total,
        'successful': successful,
        'success_rate': successful / total,
        'avg_quality_score': sum(r.get('quality_score', 0) for r in results) / total,
        'avg_facts': sum(len(r.get('structured_answer', {}).get('facts', [])) for r in results) / total,
        'avg_unknown': sum(len(r.get('structured_answer', {}).get('unknown', [])) for r in results) / total
    }

    # 2. 质量分布
    quality_bins = [0, 0.6, 0.8, 0.9, 1.0]
    quality_counts = {
        '< 0.6 (差)': 0,
        '0.6-0.8 (一般)': 0,
        '0.8-0.9 (良好)': 0,
        '0.9-1.0 (优秀)': 0
    }

    for r in results:
        score = r.get('quality_score', 0)
        if score < 0.6:
            quality_counts['< 0.6 (差)'] += 1
        elif score < 0.8:
            quality_counts['0.6-0.8 (一般)'] += 1
        elif score < 0.9:
            quality_counts['0.8-0.9 (良好)'] += 1
        else:
            quality_counts['0.9-1.0 (优秀)'] += 1

    report['quality_distribution'] = quality_counts

    # 3. 人工检查结果（如果有）
    if manual_review:
        report['manual_review'] = manual_review

    # 4. 建议
    if report['overview']['success_rate'] < 0.95:
        report['recommendations'].append('成功率较低，检查API调用和检索质量')

    if report['overview']['avg_quality_score'] < 0.8:
        report['recommendations'].append('平均质量分较低，考虑优化Prompt或检索')

    if quality_counts['< 0.6 (差)'] > total * 0.1:
        report['recommendations'].append(f"低质量样本过多({quality_counts['< 0.6 (差)']}个)，需要重新生成")

    return report


def print_final_report(report: Dict):
    """打印最终报告"""

    print(f"\n{'='*60}")
    print(f"Reference Answer 质量总报告")
    print(f"{'='*60}")

    print(f"\n📊 概览:")
    overview = report['overview']
    print(f"  总问题数: {overview['total_questions']}")
    print(f"  成功生成: {overview['successful']} ({overview['success_rate']:.1%})")
    print(f"  平均质量分: {overview['avg_quality_score']:.2f}")
    print(f"  平均Facts数: {overview['avg_facts']:.1f}")
    print(f"  平均Unknown数: {overview['avg_unknown']:.1f}")

    print(f"\n📈 质量分布:")
    for category, count in report['quality_distribution'].items():
        pct = count / overview['total_questions'] * 100
        bar = '█' * int(pct / 2)
        print(f"  {category:20s} {count:4d} ({pct:5.1f}%) {bar}")

    if 'manual_review' in report:
        print(f"\n👤 人工检查:")
        manual = report['manual_review']
        print(f"  检查样本数: {manual.get('total_reviewed', 'N/A')}")
        print(f"  通过率: {manual.get('pass_rate', 'N/A')}")

    print(f"\n💡 建议:")
    if report['recommendations']:
        for i, rec in enumerate(report['recommendations'], 1):
            print(f"  {i}. {rec}")
    else:
        print(f"  ✅ 质量良好，无特别建议")

    print(f"\n{'='*60}")

if __name__ == "__main__":
```
